    "name": "example",
    "version": "0.1",
    "description": "Demo plugin: flag responses larger than 1 MB.",
}

def scanner_rules():
    from weblore.scanner import Finding

    def big_responses(ctx):
        if len(ctx.resp_body) > 1_000_000:
            yield Finding(
                severity="info",
                title="Very large response body",
                host=ctx.host, url=ctx.url, request_id=ctx.history_id,
                evidence=f"{len(ctx.resp_body)} bytes",
            )

    return [big_responses]
